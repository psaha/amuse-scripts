    \begin{frame}{Local Neighbourhood}
    	\begin{columns}[c]
    		\column{0.65\textwidth}    		
    		\begin{itemize}
    			\item Properties of star using	\textbf{Local Neighbourhood} of particles.    			
    			\item \textbf{Smoothing Length} $h$ sets the size of it.   			
    			\item Kernel distributes the contribution of \textbf{each particle smoothly}:
    			\begin{equation*}
    				W(r,h)
    				=
    				\frac{1}{\pi^{3/2}h^3}
    				\exp\left(-\frac{r^2}{h^2}\right).
    			\end{equation*}
    			\begin{equation*}
    				\int W(r,h)\,d^3r = 1.
    			\end{equation*}
    			\item Particles closer to the point of interest contribute
    			\textbf{more strongly}.
    		\end{itemize}   		
    		\column{0.35\textwidth}
    		\begin{figure}
    			\centering
    			\includegraphics[width=\textwidth]{fig/Gauss_Kernal.jpg}
    			\caption*{\tiny \textbf{Gaussian smoothing kernel}}
    		\end{figure}
    	\end{columns}
    \end{frame}

    \begin{frame}{Density and Acceleration}
    	\begin{columns}[c]
    		\column{0.65\textwidth}    		
    		\begin{itemize}    			
    			\item \textbf{Density} using the smoothing kernel:    			
    			\begin{equation*}
    				\rho_i =
    				\sum_j m_j\,W(\mathbf{r}_i-\mathbf{r}_j,h).
    			\end{equation*}    			   			
    			\item Pressure gradient produce the \textbf{hydrodynamic acceleration}:    			
    			\begin{equation*}
    				\mathbf{a}_{i}^{\,\mathrm{P}}
    				=
    				-\sum_j m_j
    				\left(
    				\frac{P_i}{\rho_i^2}
    				+
    				\frac{P_j}{\rho_j^2}
    				\right)
    				\nabla_i W_{ij}.
    			\end{equation*}    			
    			\item The \textbf{particle acceleration} is then determined from
    			the combined forces:    			
    			\begin{equation*}
    				\boxed{
    					\mathbf{a}_i =
    					\mathbf{a}_{i}^{\,\mathrm{P}}
    					+
    					\mathbf{a}_{i}^{\,\mathrm{G}}
    					-
    					\nu\mathbf{v}_i
    				}
    			\end{equation*}    			
    		\end{itemize}
    		\column{0.35\textwidth}   		
    		\begin{figure}
    			\centering
    			\includegraphics[width=\textwidth]{fig/polytrope_profile.png}    			
    			\caption*{\tiny	\textbf{From local neighbours to Density}}
    		\end{figure}   		
    	\end{columns}
    \end{frame}

    \begin{frame}{Equilibrium: The Balancing Force}   	
    	\begin{columns}[c]    		
    		\column{0.65\textwidth}    		
    		\begin{itemize}
    			\item Equilibrium - \textbf{pressure balances gravity}.    			
    			\item The pressure gradient provides an outward force:
    			\[
    			\mathbf{a}_{P}
    			=-\frac{1}{\rho}\nabla P
    			\]    			
    			\item Gravity provides an inward acceleration:
    			\[
    			\mathbf{a}_{G}=-\nabla\Phi
    			\]   			
    			\item At equilibrium:
    			\[
    			\boxed{\mathbf{a}_{P}+\mathbf{a}_{G}=0}
    			\]
    		\end{itemize}    		
    		\column{0.35\textwidth}    		
    		\centering
    		\begin{figure}
    			\centering
    			\animategraphics[autoplay, loop, width=\linewidth]{5}{fig/frames_star/frame}{1}{60}
    			\caption*{\tiny \textbf{Reaching in Equilibrium}}
    		\end{figure}    		
    	\end{columns}    	
    \end{frame}    

    \begin{frame}{Evolution in Acceleration}	
    	\begin{columns}[c]		
		    \column{0.6\textwidth}		
		    \begin{itemize}
			    \item Particles are evolved using the \textbf{leap-frog method}.			
		       	\item \textbf{Velocity} is updated by a half-step:
			    \begin{equation*}
			    	\mathbf{v}_i^{n+1/2} = 	\mathbf{v}_i^n + \frac{\Delta t}{2}\mathbf{a}_i^n
			    \end{equation*}			
		    	\item \textbf{Position} is updated by a full step:
			    \begin{equation*}
			    	\mathbf{r}_i^{n+1} = \mathbf{r}_i^n	+ \Delta t\,\mathbf{v}_i^{n+1/2}
			    \end{equation*}			
			    \item \textbf{Acceleration} is then recomputed for the next step.
	    	\end{itemize}		
		    \column{0.4\textwidth}		
		    \centering
		    {\color{blue}\Large\textbf{Leap-Frog}}		
		    \vspace{0.3cm}		
		    \begin{tikzpicture}[>=stealth, scale=0.8, every node/.style={font=\small}]			
		    	\draw[->, thick] (0,0) -- (5,0) node[right] {$t$};			
			    \fill[red] (0.8,0) circle (3pt);
	    		\fill[red] (4.2,0) circle (3pt);			
	    		\node[red, above] at (0.8,0) {$\mathbf r_i^n$};
    			\node[red, above] at (4.2,0) {$\mathbf r_i^{n+1}$};			
    			\draw[blue, thick, ->] (2.5,0.2) -- (2.5,1.1);			
	    		\node[blue, above] at (2.5,1.1)
	    		{$\mathbf v_i^{n+1/2}$};			
	    		\draw[red, thick, ->] (1,-0.35) -- (4,-0.35);			
	    		\node[red, below] at (2.5,-0.35)
	    		{\scriptsize $\Delta t$};			
	    	\end{tikzpicture}		
		    \vspace{0.3cm}		
		    \begin{block}{Key idea}
		    	\centering
		    	{\color{red}$\mathbf r$: full steps}
	    		\qquad
	    		{\color{blue}$\mathbf v$: half steps}
	    	\end{block}		
    	\end{columns}	
    \end{frame}

    \begin{frame}{Binary System with SPH Particles}   	
    	\begin{itemize}
    		\item Each star is represented by a collection of SPH particles, which carries:
    		\[
    		m_i,\quad \mathbf r_i,\quad \mathbf v_i,\quad
    		\rho_i,\quad u_i,\quad h_i
    		\]   		
    		\item The particles interact through:
    		\begin{itemize}
    			\item gravitational force,
    			\item pressure force,
    			\item hydrodynamic interactions.
    		\end{itemize}
    	\end{itemize}
    	
    	\begin{block}{Advantage}
    		\centering
    		The stellar object can be described analytically, while SPH provides a natural representation of the stellar surface and its complex geometry.
    	\end{block}    	
    \end{frame}

    \begin{frame}{Why a Close Binary Star System?}
    	
    	\begin{itemize}
    		\item The gravitational field is \textbf{no longer spherically symmetric}.    		
    		\item The stars can undergo \textbf{strong tidal interaction}.    		
    		\item Their \textbf{surfaces can become distorted} from spherical symmetry.    		
    		\item The observed \textbf{brightness distribution changes} with orbital phase.
    	\end{itemize}
    	
    	\begin{block}{Key Idea}
    		\centering
    		A binary system requires the \textbf{dynamics and surface structure} to be evolved together.
    	\end{block}
       	
    \end{frame}
